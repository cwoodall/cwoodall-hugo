{{- /* Markdown image render hook: ![alt text](photo.jpg "Optional caption") - If the path is a page-bundle resource (relative path next to index.md), emit responsive webp variants (srcset), width/height, lazy loading, and link to a large version. Resizing happens at build time; originals are never served. - Otherwise (/img/... in static/, or a remote URL) emit a plain , so older posts keep working unchanged. - A markdown title becomes a visible caption. Output uses s because goldmark wraps images in

; see _figures.scss. */ -}} {{- $src := .Destination -}} {{- $res := false -}} {{- if not (or (hasPrefix $src "/") (in $src "://")) -}} {{- $res = .Page.Resources.GetMatch $src -}} {{- end -}} {{- if and $res (eq $res.ResourceType "image") -}} {{- $variants := slice -}} {{- $widths := slice -}} {{- range slice 480 800 1200 1600 -}} {{- $r := partial "image-resize.html" (dict "img" $res "width" .) -}} {{- if not (in $widths $r.Width) -}} {{- $variants = $variants | append (printf "%s %dw" $r.RelPermalink $r.Width) -}} {{- $widths = $widths | append $r.Width -}} {{- end -}} {{- end -}} {{- $img := partial "image-resize.html" (dict "img" $res "width" 1200) -}} {{- $full := partial "image-resize.html" (dict "img" $res "width" 2000) -}} {{- else -}} {{- if not (or (hasPrefix $src "/") (in $src "://")) -}} {{- errorf "render-image: %q not found in page bundle of %s" $src .Page.File.Path -}} {{- end -}} {{- end -}} {{- with .Title }}{{ . | markdownify }}{{ end -}} {{- /* trim */ -}}
