{{/* Resize an image resource to at most `width` px wide as webp, rotating it upright from its EXIF orientation first. Never upscales; returns the original for gif/svg (keeps animation / vectors). Usage: partial "image-resize.html" (dict "img" $res "width" 800) */}} {{ $img := .img }} {{ $out := $img }} {{ if not (in (slice "image/gif" "image/svg+xml") $img.MediaType.Type) }} {{ $upright := $img.Filter images.AutoOrient }} {{ $w := .width }} {{ if gt $w $upright.Width }}{{ $w = $upright.Width }}{{ end }} {{ $out = $upright.Resize (printf "%dx webp q80" $w) }} {{ end }} {{ return $out }}
